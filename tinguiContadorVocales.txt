tingui 'ContadorVocales';
var
  texto : string
  longitud : real
  i : real
  contador : real
  caracter : string
{
  read('Ingrese una frase para contar vocales: ', texto);
  
  longitud <- largo(texto);
  i <- 1;
  contador <- 0;
  
  While i <= longitud Do
  {
     caracter <- subcad(texto, i, 1);
     
     if caracter = 'a' then { contador <- contador + 1; };
     if caracter = 'e' then { contador <- contador + 1; };
     if caracter = 'i' then { contador <- contador + 1; };
     if caracter = 'o' then { contador <- contador + 1; };
     if caracter = 'u' then { contador <- contador + 1; };
     
     i <- i + 1;
  };
  
  write('La cantidad total de vocales es:');
  write(contador);
}.
